+# [[file:../../../org/20260425230731-org_llm.org::*tests/test_perf.py][test_perf.py:1]]
+"""Tests for pure helpers in org_llm.perf — _normalize_model + _row_tok_s."""
+from __future__ import annotations
+
+from org_llm.perf import _normalize_model, _row_tok_s
+
+
+# ── _normalize_model ───────────────────────────────────────────────────────────
+
+def test_normalize_strips_latest_suffix():
+    assert _normalize_model("phi3.5:latest") == "phi3.5"
+
+
+def test_normalize_strips_latest_case_insensitive():
+    assert _normalize_model("PHI3.5:LATEST") == "phi3.5"
+
+
+def test_normalize_lowercases():
+    assert _normalize_model("Llama3.3") == "llama3.3"
+
+
+def test_normalize_preserves_non_latest_variant():
+    assert _normalize_model("deepseek-r1:7b") == "deepseek-r1:7b"
+
+
+def test_normalize_empty_string():
+    assert _normalize_model("") == ""
+
+
+def test_normalize_none():
+    assert _normalize_model(None) == ""
+
+
+def test_normalize_strips_surrounding_whitespace():
+    assert _normalize_model("  phi3.5:latest  ") == "phi3.5"
+
+
+def test_normalize_idempotent():
+    name = "nomic-embed-text:latest"
+    once = _normalize_model(name)
+    twice = _normalize_model(once)
+    assert once == twice
+
+
+# ── _row_tok_s ─────────────────────────────────────────────────────────────────
+
+# Helper: a response long enough and a duration slow enough to produce a reading
+_VALID_RESPONSE = "a" * 40   # 40 chars → 10 tok at 4 chars/tok
+_VALID_DURATION = 4000       # 4 s → 2.5 tok/s expected
+
+
+def test_row_tok_s_happy_path():
+    result = _row_tok_s(_VALID_RESPONSE, _VALID_DURATION)
+    assert result is not None
+    assert abs(result - 2.5) < 0.01
+
+
+def test_row_tok_s_none_on_error_outcome():
+    assert _row_tok_s(_VALID_RESPONSE, _VALID_DURATION, outcome="error") is None
+
+
+def test_row_tok_s_none_on_error_outcome_case_insensitive():
+    assert _row_tok_s(_VALID_RESPONSE, _VALID_DURATION, outcome="ERROR") is None
+
+
+def test_row_tok_s_none_on_none_response():
+    assert _row_tok_s(None, _VALID_DURATION) is None
+
+
+def test_row_tok_s_none_on_empty_response():
+    assert _row_tok_s("", _VALID_DURATION) is None
+
+
+def test_row_tok_s_none_on_none_duration():
+    assert _row_tok_s(_VALID_RESPONSE, None) is None
+
+
+def test_row_tok_s_none_on_zero_duration():
+    assert _row_tok_s(_VALID_RESPONSE, 0) is None
+
+
+def test_row_tok_s_none_on_sub_second_duration():
+    # Anything under 1000 ms is treated as cache-hit / error, not real generation
+    assert _row_tok_s(_VALID_RESPONSE, 999) is None
+
+
+def test_row_tok_s_none_on_short_response():
+    # Fewer than 10 chars → too noisy to estimate
+    assert _row_tok_s("a" * 9, _VALID_DURATION) is None
+
+
+def test_row_tok_s_none_on_responseerror_prefix():
+    msg = "ResponseError: model requires more system memory (6.9 GiB) blah blah"
+    assert _row_tok_s(msg, _VALID_DURATION) is None
+
+
+def test_row_tok_s_none_on_http_prefix():
+    msg = "HTTP 500 Internal Server Error — something went wrong on the server side"
+    assert _row_tok_s(msg, _VALID_DURATION) is None
+
+
+def test_row_tok_s_none_on_traceback_prefix():
+    msg = "Traceback (most recent call last):\n  File ...\nValueError: oops"
+    assert _row_tok_s(msg, _VALID_DURATION) is None
+
+
+def test_row_tok_s_none_on_status_code_5xx():
+    msg = "status code: 500 server error occurred here"
+    assert _row_tok_s(msg, _VALID_DURATION) is None
