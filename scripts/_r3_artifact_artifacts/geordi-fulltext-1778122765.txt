3	
4	Derives tok/s for every model the user has actually run by mining the
5	existing `History` table (kind=llm rows have model + duration_ms +
6	response). No new schema — everything we need is already logged.
7	
8	Three jobs:
9	
10	1. ``recent_tok_s(model)`` — rolling-average tok/s for one model based
11	   on the last N completed chat calls. Used by the lag detector to
12	   know what "normal" looks like for this model.
13	
14	2. ``fastest_known(role, current=...)`` — across all models we have
15	   history for, pick the fastest one that supports ``role``. Used by
16	   ``models --upgrade`` and the lag-detector hint to point users at a
17	   known-faster alternative without re-running the benchmark.
18	
19	3. ``check_lag(model, elapsed_s, response_chars)`` — called inline
20	   from ``llm.chat()``. Returns a one-line warning string if the call
21	   was meaningfully slower than this model's recent baseline AND a
22	   faster alternative is on disk; returns None otherwise. Cheap.

25	writes one row to History via ``logbook.track_event``. Adding a parallel

65	def _row_tok_s(response: str | None, duration_ms: int | None,
66	                 outcome: str | None = None) -> float | None:
67	    """tok/s for a single History row, or None if it can't be measured.
68	
69	    Filters out junk samples that would otherwise inflate baselines:
70	
71	      • Error rows. The History.response column stores the exception
72	        message text on outcome=error (e.g. "ResponseError: model
73	        requires more system memory (6.9 GiB)…"). Treating those as
74	        real generation made fastest_known() report mistral-nemo at
75	        ~28 tok/s when it actually OOMs and runs ~1 tok/s — every
76	        "fast" mistral row was a fast-failing 500 with the error
77	        message as the response.
78	
79	      • Sub-second durations. A real 10+ chars completion at 28+
80	        tok/s is conceivable on a fast cloud endpoint but on local
81	        CPU is almost always a cache hit / refusal / immediate
82	        error, not actual generation work.
83	
84	      • Tiny replies. Single-word answers / refusals give noisy
85	        per-call tok/s estimates.
86	    """
87	    if outcome and outcome.lower() == "error":
88	        return None
89	    if not response or not duration_ms or duration_ms <= 0:
90	        return None
91	    if duration_ms < 1000:
92	        # Anything that "completed" in under a second on local
93	        # hardware is almost certainly a cache hit or error response.
94	        return None
95	    chars = len(response)
96	    if chars < 10:
97	        return None
98	    # Defensive: error messages routinely start with "ResponseError:"
99	    # or "HTTP" / "401" / "404" patterns. Drop any response that
100	    # looks like a stringified exception. This catches rows where
101	    # outcome was incorrectly recorded as "ok" but the body is an
102	    # error trail.
103	    head = response.lstrip()[:60].lower()
104	    if (head.startswith(("responseerror", "http", "error:",
105	                            "traceback", "exception"))
106	            or "status code: 5" in head or "status code: 4" in head):
107	        return None
108	    tok_est = chars / _CHARS_PER_TOKEN
109	    return tok_est / (duration_ms / 1000.0)

114	def recent_tok_s(model: str, *, limit: int = 20) -> float | None:
115	    """Average tok/s for the last `limit` completed chat calls to `model`.
116	
117	    Returns None when we don't have enough samples yet.
118	    """
119	    eng = _engine()
120	    if eng is None:
121	        return None
122	    from .db import History, get_session
123	    target = _normalize_model(model)
124	    samples: list[float] = []
125	    with get_session(eng) as s:
126	        rows = (s.query(History)
127	                  .filter(History.kind == "llm",
128	                          History.duration_ms.isnot(None),
129	                          History.response.isnot(None))
130	                  .order_by(History.id.desc())
131	                  .limit(limit * 4).all())  # over-fetch; filter below
132	    for r in rows:
133	        if _normalize_model(r.model or "") != target:
134	            continue
135	        v = _row_tok_s(r.response, r.duration_ms, getattr(r, "outcome", None))
136	        if v is None:
137	            continue
138	        samples.append(v)
139	        if len(samples) >= limit:
140	            break
141	    if len(samples) < 3:
142	        return None
143	    return sum(samples) / len(samples)

155	def fastest_known(role: str, *, current_model: str = "",
156	                    limit_per_model: int = 20) -> _Alt | None:
157	    """Across every model with chat history, find the fastest one that
158	    supports `role` per CATALOG (or has no catalog entry — then we
159	    assume "chat", since plain chat is the default role tag).
160	
161	    Returns None when no alternative exists or no model has data.
162	    """
163	    eng = _engine()
164	    if eng is None:
165	        return None
166	    from .db import History, get_session
167	    from .models import CATALOG, _quality
168	    role_models: dict[str, list[float]] = {}
169	    with get_session(eng) as s:
170	        rows = (s.query(History)
171	                  .filter(History.kind == "llm",
172	                          History.duration_ms.isnot(None),
173	                          History.response.isnot(None))
174	                  .order_by(History.id.desc())
175	                  .limit(limit_per_model * 50).all())
176	    for r in rows:
177	        m = _normalize_model(r.model or "")
178	        if not m:
179	            continue
180	        v = _row_tok_s(r.response, r.duration_ms, getattr(r, "outcome", None))
181	        if v is None:
182	            continue
183	        role_models.setdefault(m, []).append(v)
184	        if len(role_models[m]) > limit_per_model * 2:
185	            role_models[m] = role_models[m][:limit_per_model]
186	
187	    cur = _normalize_model(current_model)
188	    best: _Alt | None = None
189	    for m, samples in role_models.items():
190	        if m == cur:
191	            continue
192	        if len(samples) < 3:
193	            continue
194	        # Match catalog entry by stem to confirm role support
195	        meta = next((c for c in CATALOG
196	                      if c.tag.split(":")[0] == m.split(":")[0]),
197	                     None)
198	        roles = meta.roles if meta else ("chat",)
199	        if role and role not in roles:
200	            continue
201	        tok_s = sum(samples) / len(samples)
202	        qual = _quality(m) or 50
203	        cand = _Alt(model=m, tok_s=tok_s, n=len(samples), quality=qual)
204	        if best is None or cand.tok_s > best.tok_s:
205	            best = cand
206	    return best

219	def check_lag(model: str, elapsed_s: float, response: str,
220	               *, slow_factor: float = 3.0,
221	               min_elapsed: float = 8.0) -> LagWarning | None:
222	    """Inline lag check — call from inside the chat path right after
223	    a request returns. Returns a `LagWarning` if (a) the call took
224	    more than `min_elapsed` seconds AND (b) we have a faster
225	    alternative already on disk.
226	
227	    Cheap: one indexed History query (last ~20 rows for this model)
228	    plus one query for fastest-known. Skip when DB is missing or
229	    samples are insufficient.
230	
231	    `slow_factor` tunes when "current call slower than baseline" is
232	    noisy enough to mention. 3× is conservative — first-token latency
233	    dominates short calls so we don't want to alarm on noise.
234	    """
235	    if elapsed_s < min_elapsed:
236	        return None
237	    cur_tok = _row_tok_s(response, int(elapsed_s * 1000))
238	    baseline = recent_tok_s(model)
239	    # Only escalate if we KNOW it's slow. Without a baseline (cold
240	    # start), still warn when a faster pulled alternative exists —
241	    # the user benefits even on the first call.
242	    if baseline and cur_tok and cur_tok * slow_factor > baseline:
243	        return None  # within normal noise
244	    alt = fastest_known("chat", current_model=model)
245	    if alt is None:
246	        return None
247	    if cur_tok and alt.tok_s < cur_tok * 1.3:
248	        # Alternative isn't meaningfully faster; not worth nagging.
249	        return None
250	    return LagWarning(
251	        elapsed_s        = elapsed_s,
252	        baseline_tok_s   = baseline,
253	        current_tok_s    = cur_tok,
254	        suggested_model  = alt.model,
255	        suggested_tok_s  = alt.tok_s,
256	    )

294	            if not isinstance(existing, list):

296	        except Exception:
297	            existing = []

299	            "ts":              time.time(),

305	            "suggested_model": suggested_model,
306	            "suggested_tok_s": round(suggested_tok_s, 2),
307	        })
308	        # Cap the buffer so it doesn't grow unbounded.
309	        existing = existing[-_ALERT_BUFFER_LIMIT:]

346	def regression_warning(model: str, *, recent_window: int = 5,
347	                         baseline_window: int = 50,
348	                         slow_factor: float = 1.5) -> str | None:
349	    """Compare recent runs for `model` against an older baseline.
350	    Returns a one-line warning string when the model has gotten
351	    materially slower vs its history (e.g. ollama upgrade, RAM
352	    pressure from a new long-running process).
353	    """
354	    eng = _engine()
355	    if eng is None:
356	        return None
357	    from .db import History, get_session
358	    target = _normalize_model(model)
359	    recent: list[float] = []
360	    older:  list[float] = []
361	    with get_session(eng) as s:
362	        rows = (s.query(History)
363	                  .filter(History.kind == "llm",
364	                          History.duration_ms.isnot(None),
365	                          History.response.isnot(None))
366	                  .order_by(History.id.desc())
367	                  .limit(baseline_window * 6).all())
368	    for r in rows:
369	        if _normalize_model(r.model or "") != target:
370	            continue
371	        v = _row_tok_s(r.response, r.duration_ms, getattr(r, "outcome", None))
372	        if v is None:
373	            continue
374	        if len(recent) < recent_window:
375	            recent.append(v)
376	        elif len(older) < baseline_window:
377	            older.append(v)
378	        else:
379	            break
380	    if len(recent) < 3 or len(older) < 5:
381	        return None
382	    r_avg = sum(recent) / len(recent)
383	    o_avg = sum(older)  / len(older)
384	    if r_avg <= 0 or o_avg <= 0:
385	        return None
386	    if o_avg > r_avg * slow_factor:
387	        slowdown = o_avg / r_avg
388	        return (f"{model}: recent runs are {slowdown:.1f}× slower "
389	                f"than baseline ({r_avg:.1f} vs {o_avg:.1f} tok/s). "
390	                f"Run [bold]org-llm doctor --power-boost[/bold] or "
391	                f"[bold]org-llm models --upgrade[/bold].")
392	    return None
