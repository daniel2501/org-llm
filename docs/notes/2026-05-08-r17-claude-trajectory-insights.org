# scripts/_r18_trajectory_replay.py (new file, not yet written)

TRAJECTORIES = {
    "B1":  [("read_file",  {"path": "{target_file}"}),
            *[("edit_file", {"path": "{target_file}",
                              "old_template": "prose_around_{candidate}",
                              "new_template": "[[id:{uuid}][{label}]]-wrapped"})
              for _ in range(5)]],
    "B5":  [("read_file",  {"path": "{target_file}"}),
            ("edit_file",  {"path": "{target_file}",
                              "old_template": "{prefetch.longest_paragraph_text}",
                              "new_template": "{tightened_30pct}"})],
    "B7":  [("read_file",  {"path": "org_llm/avatars.py"}),
            ("noop_if",    {"predicate": "file_already_typed"})],
    "B11": [("read_tail",  {"path": "{target_file}", "n": 120}),
            ("run_python", {"code": "import uuid; print(uuid.uuid4())"}),
            ("grep",       {"pattern": "^\\*\\* DEC-0", "path": "{target_file}",
                             "post": "tail -5"}),
            ("read_file",  {"path": "{target_file}",
                             "offset": "{last_dec_line}", "limit": 30}),
            ("shell",      {"cmd": "tail -5 {target_file}"}),
            ("grep",       {"pattern": ":STATUS:", "path": "{target_file}",
                             "post": "sort -u"}),
            ("edit_file",  {"path": "{target_file}",
                             "old_template": "{tail_5_lines}",
                             "new_template": "{tail_5_lines}\\n{new_dec_skeleton}"})],
    "B13": [("read_file",  {"path": "LICENSE"}),
            ("edit_file",  {"path": "LICENSE",
                              "old_template": "Copyright (c) 2025 {holder}",
                              "new_template": "Copyright (c) 2026 {holder}"})],
    "B20": [("noop_if",    {"predicate": "prefetch.has_summary"})],
    "B25": [("write_file", {"path": "{target_file}",
                              "content_template": "{shout_defun_body}"}),
            ("eval_elisp", {"code": "(progn (load \"{abs_path}\") "
                                     "(org-llm-r17-shout \"hello\"))",
                              "expect_substring": "HELLO!!!"})],
    "B26": [("read_file",       {"path": "{target_file}"}),
            ("edit_file",       {"path": "{target_file}",
                                   "old_template": "{anchor_defcustom_block}",
                                   "new_template": "{anchor + new_defcustom}"}),
            ("load_elisp_file", {"path": "{target_file}"})],
}

def replay_claude_trajectory(task, variant, prefetch, target_file, llm_call):
    """Walk the canonical trajectory; LLM only fills templates, harness drives.

    Each step is one of:
      - mechanical (read_file, run_python, grep, shell): harness runs directly
      - llm-bound (edit_file, write_file): variant LLM is asked ONLY to fill
        the {…} templates given the read/grep results so far.

    The variant never decides WHICH tool to call next — only what content to
    put in the slots. Survey is mechanical; insertion is LLM-shaped.
    """
    ctx = {"prefetch": prefetch, "target_file": target_file,
           "step_results": []}
    for tool, args in TRAJECTORIES[task]:
        if tool == "noop_if" and eval_predicate(args["predicate"], ctx):
            return {"status": "noop", "reason": args["predicate"]}
        if tool in ("read_file", "run_python", "grep", "shell",
                     "load_elisp_file", "eval_elisp", "read_tail"):
            result = HARNESS_TOOLS[tool](**resolve(args, ctx))
            assert success_check(tool, result), f"{tool} failed: {result}"
            ctx["step_results"].append((tool, result))
        elif tool in ("edit_file", "write_file"):
            # LLM fills {…} placeholders given accumulated context
            filled = llm_call(variant=variant,
                                tool=tool,
                                templates=args,
                                context=ctx)
            result = HARNESS_TOOLS[tool](**filled)
            ctx["step_results"].append((tool, result))
    return {"status": "ok", "trace": ctx["step_results"]}
#+end_src

* Implementation notes — easy vs. needs-refactor
| Task | Easy / hard | Why                                                                              |
|------+-------------+----------------------------------------------------------------------------------|
| B13  | trivial     | 1 read + 1 edit; no shell-shaped survey                                          |
| B20  | trivial     | pure no-op gated on prefetch                                                     |
| B7   | trivial     | 1 read + no-op decision predicate                                                |
| B5   | easy        | 1 read + 1 edit; LLM only fills the tightened-text slot                          |
| B26  | easy        | read + edit + load_elisp_file (already in FOSS surface)                          |
| B25  | easy        | write_file + eval_elisp (collapse Claude's emacs --batch into FOSS eval_elisp)   |
| B1   | medium      | 5 stacked edits; LLM must pick which 5 candidates from prefetch list             |
| B11  | hard        | needs constrained survey_shell tool (tail/grep/wc/sort -u) OR redesign as repeated read_file w/ offsets + grep-equivalent (e.g. find_canonical_id-style) |

Hard part for B11: 3 of 8 steps are read-only shell calls. Two options:
1. Add a =survey_shell= tool whitelisting {head, tail, grep, wc, sort -u}.
   Cleanest replay; small new tool surface.
2. Pre-compute the survey results in prefetch (last_dec_line, dec_numbering,
   allowed_status_values, last_5_lines). Move the work upstream — variant
   sees a richer prefetch and the trajectory shrinks to {read_file,
   run_python(uuid), edit_file}. Aligns with R18 design's "manager
   pre-fetch + agent narrates" pattern.

Recommendation: option (2). It is aligned with the existing
=manager_prefetch_pattern= memory, doesn't widen the tool surface, and makes
the trajectory definition cleaner (mechanical survey → LLM-only-fills-slots).
